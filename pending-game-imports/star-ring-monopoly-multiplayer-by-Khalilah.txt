    const $ = id => document.getElementById(id);
    const money = n => "¥" + Math.round(n);
    const wait = ms => new Promise(resolve => setTimeout(resolve, ms));
    const alivePlayers = () => state.players.filter(p => !p.bankrupt);
    const currentPlayer = () => state.players[state.currentIndex];

    const propCount = p => p.props.length;

    function escapeHtml(s) {
      return String(s).replace(/[&<>"']/g, m => ({ "&":"&amp;", "<":"&lt;", ">":"&gt;", '"':"&quot;", "'":"&#39;" }[m]));
    }

    function toast(text) {
      $("toast").textContent = text;
      $("toast").classList.add("show");
      setTimeout(() => $("toast").classList.remove("show"), 1800);
    }

    function addLog(text) {
      state.log.unshift(text);
      state.log = state.log.slice(0, 34);
      $("log").innerHTML = state.log.map(item => `<div>${escapeHtml(item)}</div>`).join("");
    }

    function specialLabel(cell) {
      if (cell.type === "chance") return "Event";
      if (cell.type === "tax") return "-" + money(cell.amount);
      if (cell.type === "bonus") return "+" + money(cell.amount);
      return "";
    }

    function initBoard() {
      const grid = $("grid");
      board.forEach((cell, i) => {
        const [col, row] = path[i];
        const el = document.createElement("div");
        el.className = "cell" + (cell.type === "start" || i === 19 ? " corner" : "");
        el.style.gridColumn = String(col + 1);
        el.style.gridRow = String(row + 1);
        el.innerHTML = `
          <div class="owner" id="owner-${i}"></div>
          <span class="emoji">${cell.emoji}</span>
          <div class="name">${cell.name}</div>
          <div class="price">${cell.price ? money(cell.price) : specialLabel(cell)}</div>
          <div class="token p0 hidden" id="token-0-${i}"></div>
          <div class="token p1 hidden" id="token-1-${i}"></div>
          <div class="token p2 hidden" id="token-2-${i}"></div>
          <div class="token p3 hidden" id="token-3-${i}"></div>
        `;
        grid.appendChild(el);
      });
    }

    function render() {
      $("stats").innerHTML = state.players.length ? state.players.map((p, i) => `
        <div class="stat">
          <strong>${["💙","🌹","🌿","💜"][i]} ${escapeHtml(p.name)}${p.bankrupt ? " (out)" : ""}</strong>
          <span>Cash ${money(p.money)} | Properties ${propCount(p)} | Assets ${money(asset(p))}</span>
        </div>
      `).join("") : `<div class="stat"><strong>Waiting to begin</strong><span>Choose your characters</span></div>`;

      const p = currentPlayer();
      $("turnPill").textContent = state.ended ? "Finished" : p ? `Round ${state.round}/${state.maxRounds} | ${p.name}` : "Getting ready";
      $("rollBtn").disabled = state.ended || !p || p.type !== "human" || !!state.awaitingChoice;
      $("saveBtn").disabled = !state.players.length || state.ended;

      board.forEach((_, i) => {
        for (let pi = 0; pi < 4; pi++) {
          const token = $("token-" + pi + "-" + i);
          const player = state.players[pi];
          if (token) token.classList.toggle("hidden", !player || player.pos !== i || player.bankrupt);
        }
        const ownerIndex = state.owners[i];
        $("owner-" + i).className = "owner" + (ownerIndex !== null && ownerIndex !== undefined ? " p" + ownerIndex : "");
      });
    }

    async function initHost() {
      initBoard();
      addLog("✨ Welcome to opening night.");

      try {
        if (API && API.setTitleBar) {
          API.setTitleBar({
            material: "glass",
            background: "rgba(10,15,32,.45)",
            color: "#fffaf0",
            borderColor: "rgba(255,255,255,.12)",
            buttonBackground: "rgba(255,255,255,.10)",
            buttonColor: "#fffaf0"
          });
        }
        if (API && API.getPlayerProfile) state.playerProfile = await API.getPlayerProfile();
        if (API && API.listAvailableCharacters) state.characters = await API.listAvailableCharacters();
        await detectSave();
      } catch (e) {
        addLog("Host features unavailable - you can play locally instead.");
      }

      renderRoles();
      render();
    }

    async function detectSave() {
      if (!API || !API.loadGame) return;
      const save = await API.loadGame();
      if (!save || save.version !== 5 || save.ended) return;

      state.hasSave = true;
      state.savedGame = save;
      $("continueBtn").classList.remove("hidden");
      $("startTitle").textContent = "Saved game found";
      $("startDesc").textContent = "Pick up the last table, or choose new characters and start again.";
    }

    function renderRoles() {
      const roles = $("roles");
      if (!state.characters.length) {
        roles.innerHTML = `<p style="margin:0;">No characters loaded yet. You can try the local ones for now; opening this from the phone's game hall will list your own.</p>`;
        return;
      }

      roles.innerHTML = "";
      state.characters.forEach(c => {
        const selected = state.selectedCharacters.some(x => x.id === c.id);
        const btn = document.createElement("button");
        btn.className = "role" + (selected ? " selected" : "");
        btn.innerHTML = `
          ${c.avatar ? `<img class="avatar" src="${escapeHtml(c.avatar)}" alt="">` : `<div class="avatar"></div>`}
          <span><b>${escapeHtml(c.name)}</b><small>${escapeHtml(c.subtitle || "Ready to join the table")}</small></span>
          <span class="pill">${selected ? "Selected" : "Select"}</span>
        `;
        btn.onclick = () => toggleCharacter(c);
        roles.appendChild(btn);
      });
    }

    function toggleCharacter(c) {
      const exists = state.selectedCharacters.some(x => x.id === c.id);
      if (exists) {
        state.selectedCharacters = state.selectedCharacters.filter(x => x.id !== c.id);
      } else {
        if (state.selectedCharacters.length >= 3) {
          toast("Three characters at most");
          return;
        }
        state.selectedCharacters.push(c);
      }

      $("startBtn").textContent = state.selectedCharacters.length
        ? `Start with ${state.selectedCharacters.length} character${state.selectedCharacters.length > 1 ? "s" : ""}`
        : "Choose characters to begin";

      renderRoles();
    }

    async function startGame(demo) {
      state.demo = demo;
      if (demo) state.selectedCharacters = demoCharacters.slice();

      if (!state.selectedCharacters.length) {
        toast("Choose at least one character");
        return;
      }

      setupPlayers();
      $("startModal").classList.add("hidden");
      $("centerTitle").textContent = "The table opens";
      $("centerText").textContent = `${state.players.map(p => p.name).join(", ")} take their seats. The Star Ring auction night begins.`;
      addLog(`🥂 ${state.players.map(p => p.name).join(", ")} take their seats.`);

      await loadRolePackages();

      for (const role of state.players.filter(p => p.type === "role")) {
        const line = await roleLine(
          role,
          "opening",
          "The game has just begun. In your own voice, and mindful of your relationships, say one opening line to the player and the other characters. Do not sound like generic game dialogue."
        );
        addLog(`${role.name}：「${line}」`);
      }

      render();
      saveGame();
      maybeAutoTurn();
    }

    function setupPlayers() {
      state.players = [{
        id: "human",
        type: "human",
        name: state.playerProfile.name || "You",
        money: 1500,
        pos: 0,
        props: [],
        bankrupt: false
      }].concat(state.selectedCharacters.map(c => ({
        id: c.id,
        character: c,
        type: "role",
        name: c.name,
        money: 1500,
        pos: 0,
        props: [],
        bankrupt: false
      })));

      state.currentIndex = 0;
      state.round = 1;
      state.owners = Array(board.length).fill(null);
      state.log = [];
      state.ended = false;
      state.awaitingChoice = null;
      state.recentLines = {};
    }

    async function loadRolePackages() {
      if (state.demo || !API || !API.getRoleLightPackage) return;
      for (const p of state.players.filter(x => x.type === "role")) {
        if (!state.rolePkgs[p.id]) {
          try { state.rolePkgs[p.id] = await API.getRoleLightPackage(p.id); } catch (e) {}
        }
      }
    }

    async function continueSave() {
      if (!state.savedGame) {
        toast("No saved game to continue");
        return;
      }

      const profile = state.playerProfile;
      const characters = state.characters;

      Object.assign(state, state.savedGame);
      state.playerProfile = state.playerProfile || profile;
      state.characters = characters.length ? characters : (state.characters || []);
      state.rolePkgs = {};
      state.recentLines = state.recentLines || {};
      state.awaitingChoice = null;
      state.hasSave = false;
      state.savedGame = null;

      $("startModal").classList.add("hidden");
      $("choiceModal").classList.add("hidden");
      $("endModal").classList.add("hidden");
      $("centerTitle").textContent = "Save restored";
      $("centerText").textContent = "The property war resumes.";
      addLog("📜 Saved game loaded.");

      await loadRolePackages();
      render();
      maybeAutoTurn();
    }

    async function rollCurrent() {
      const p = currentPlayer();
      if (!p || p.bankrupt || state.ended || state.awaitingChoice) return;

      const dice = Math.ceil(Math.random() * 6);
      $("dice").textContent = dice;
      $("centerTitle").textContent = `${p.name} rolls ${dice}`;

      await move(p, dice);
      await resolveCell(p);
      render();

      if (!state.awaitingChoice && !state.ended) nextTurn();
    }

    async function move(p, steps) {
      const old = p.pos;
      p.pos = (p.pos + steps) % board.length;

      if (p.pos < old) {
        p.money += 200;
        addLog(`🏁 ${p.name} passes Start and collects ${money(200)}.`);
      }

      const cell = board[p.pos];
      $("centerText").textContent = `${p.name} lands on ${cell.emoji} ${cell.name}.`;
      addLog(`${cell.emoji} ${p.name} moves ${steps} space${steps > 1 ? "s" : ""} to ${cell.name}.`);
      render();
      await wait(360);
    }

    async function resolveCell(p) {
      const cell = board[p.pos];
      const ownerIndex = state.owners[p.pos];

      if (cell.type === "tax") {
        p.money -= cell.amount;
        addLog(`🧾 ${p.name} pays ${money(cell.amount)} at ${cell.name}.`);

        if (p.type === "role") {
          const line = await roleLine(
            p,
            "pay tax",
            `You paid ${money(cell.amount)} at ${cell.name}. React in character, given where the game stands.`
          );
          addLog(`${p.name}：「${line}」`);
        }

        await checkBankrupt(p);
        return;
      }

      if (cell.type === "bonus" || cell.type === "start") {
        if (cell.amount) {
          p.money += cell.amount;
          addLog(`${cell.emoji} ${p.name} collects a bonus of ${money(cell.amount)}.`);

          if (p.type === "role") {
            const line = await roleLine(
              p,
              "bonus",
              `You collected ${money(cell.amount)} at ${cell.name}. React in character, given your current mood.`
            );
            addLog(`${p.name}：「${line}」`);
          }
        }
        return;
      }

      if (cell.type === "chance") {
        await chance(p);
        return;
      }

      if (!cell.price) return;

      if (ownerIndex === null || ownerIndex === undefined) {
        if (p.type === "human") {
          state.awaitingChoice = { index: p.pos };
          $("choiceTitle").textContent = `${cell.emoji} ${cell.name}`;
          $("choiceText").textContent = `Unowned property - ${money(cell.price)} to buy, ${money(cell.rent)} rent. Take it?`;
          $("buyBtn").disabled = p.money < cell.price;
          $("choiceModal").classList.remove("hidden");
        } else {
          await roleBuyDecision(p, cell);
        }
        return;
      }

      if (ownerIndex !== state.players.indexOf(p)) {
        const owner = state.players[ownerIndex];
        if (!owner || owner.bankrupt) return;

        p.money -= cell.rent;
        owner.money += cell.rent;
        addLog(`💸 ${p.name} pays ${owner.name} ${money(cell.rent)} rent for ${cell.name}.`);

        if (p.type === "role") {
          const line = await roleLine(
            p,
            "pay rent",
            `You landed on ${owner.name}'s property ${cell.name} and paid ${money(cell.rent)} in rent. React in character, mindful of your relationship with them and where the game stands.`
          );
          addLog(`${p.name}：「${line}」`);
        }

        if (owner.type === "role") {
          const line = await roleLine(
            owner,
            "collect rent",
            `${p.name} landed on your property ${cell.name} and paid you ${money(cell.rent)} in rent. Say one line in character.`
          );
          addLog(`${owner.name}：「${line}」`);
        }

        await checkBankrupt(p);
      }
    }

    async function roleBuyDecision(p, cell) {
      let buy = p.money > cell.price + 220 || (cell.rent / cell.price > .18 && p.money > cell.price + 80);
      let line = "";

      if (!state.demo && API && API.callLLM && state.rolePkgs[p.id]) {
        try {
          const others = state.players
            .filter(x => x.id !== p.id)
            .map(x => `${x.name}: cash ${money(x.money)}, ${propCount(x)} properties, assets ${money(asset(x))}`)
            .join("；");

          const recent = getRecentLines(p).join(" / ");

          const result = await API.callLLM({
            characterId: p.id,
            messages: [
              ...state.rolePkgs[p.id].messages,
              {
                role: "system",
                content:
                  "You are playing Star Ring Monopoly. Stay strictly true to your character card - personality, voice, relationships, history and memories. Avoid generic board-game filler, and do not repeat what you said recently. Return JSON only, with no explanation: {\"buy\":true or false,\"line\":\"one line in character, 20 words or fewer\"}."
              },
              {
                role: "user",
                content:
                  `You are ${p.name}. Cash ${money(p.money)}, ${propCount(p)} properties, assets ${money(asset(p))}.` +
                  `The unowned property is ${cell.name}: ${money(cell.price)} to buy, ${money(cell.rent)} rent.` +
                  `Other players: ${others || "none"}. Things you said recently: ${recent || "none"}.` +
                  "Decide whether to buy, and say one line in character."
              }
            ]
          });

          const parsed = parseJson(result.content);
          if (typeof parsed.buy === "boolean") buy = parsed.buy;
          if (parsed.line) line = cleanLine(parsed.line);
        } catch (e) {}
      }

      if (!line) {
        line = fallbackRoleLine(p, buy ? "buy" : "skip");
      }

      line = rememberLine(p, line);

      if (buy && p.money >= cell.price) {
        buyProperty(p, p.pos);
        addLog(`✨ ${p.name} buys ${cell.name}.`);
      } else {
        addLog(`🕯️ ${p.name} passes on ${cell.name}.`);
      }

      addLog(`${p.name}：「${line}」`);
    }

    async function chance(p) {
      const events = [
        { icon:"💌", text:"A mysterious dividend arrives", money:110 },
        { icon:"🥂", text:"Sponsored an interstellar soiree", money:-100 },
        { icon:"📈", text:"Investment rumours heat up", money:150 },
        { icon:"🛠️", text:"Repairs to the floating terrace", money:-120 },
        { icon:"🎁", text:"Drew a patron's reward", money:80 },
        { icon:"🛰️", text:"Paid the orbital lane toll", money:-70 }
      ];

      const ev = events[Math.floor(Math.random() * events.length)];
      p.money += ev.money;
      addLog(`${ev.icon} ${p.name} draws: ${ev.text} (${ev.money >= 0 ? "+" : ""}${money(ev.money)}).`);

      if (p.type === "role") {
        const line = await roleLine(
          p,
          "chance event",
          `You drew the event "${ev.text}", changing your funds by ${ev.money >= 0 ? "+" : ""}${money(ev.money)}. React in character, given your current mood.`
        );
        addLog(`${p.name}：「${line}」`);
      }

      await checkBankrupt(p);
    }

    async function roleLine(p, scene, prompt) {
      if (!state.demo && API && API.callLLM && state.rolePkgs[p.id]) {
        try {
          const others = state.players
            .filter(x => x.id !== p.id)
            .map(x => `${x.name}: cash ${money(x.money)}, ${propCount(x)} properties, assets ${money(asset(x))}`)
            .join("；");

          const recent = getRecentLines(p).join(" / ");

          const result = await API.callLLM({
            characterId: p.id,
            messages: [
              ...state.rolePkgs[p.id].messages,
              {
                role: "system",
                content:
                  "You are playing Star Ring Monopoly with the user and other characters. Stay strictly true to your character card - personality, voice, relationships, history and memories. Avoid generic game filler, and do not repeat what you said recently. Output one line of dialogue only: no explanation, no narration, 22 words or fewer."
              },
              {
                role: "user",
                content:
                  `Scene: ${scene}. Round ${state.round}.` +
                  `Your position: cash ${money(p.money)}, ${propCount(p)} properties, assets ${money(asset(p))}.` +
                  `Other players: ${others || "none"}. Things you said recently: ${recent || "none"}.` +
                  `Task: ${prompt}`
              }
            ]
          });

          const line = cleanLine(result.content);
          if (line) return rememberLine(p, line);
        } catch (e) {}
      }

      return rememberLine(p, fallbackRoleLine(p, scene));
    }

    function getRecentLines(p) {
      return state.recentLines[p.id] || [];
    }

    function rememberLine(p, line) {
      const cleaned = cleanLine(line) || fallbackRoleLine(p, "normal");
      const list = state.recentLines[p.id] || [];

      let finalLine = cleaned;
      if (list.includes(finalLine)) {
        finalLine = fallbackRoleLine(p, "normal");
      }

      state.recentLines[p.id] = [finalLine].concat(list.filter(x => x !== finalLine)).slice(0, 8);
      return finalLine;
    }

    function cleanLine(text) {
      return String(text || "")
        .replace(/^["“”'「『]+|["“”'」』]+$/g, "")
        .replace(/^(台词|Line)[:：]\s*/i, "")
        .replace(/\s+/g, " ")
        .trim()
        .slice(0, 160);
    }

    function fallbackRoleLine(p, scene) {
      const name = p.name || "";

      const elegant = [
        "That move deserves to be remembered under these lights.",
        "Chips do speak. Just very quietly.",
        "I prefer my victories elegant.",
        "Patience. The night has not finished betting.",
        "A lovely position - and only just blooming."
      ];

      const bold = [
        "A good hand deserves everything you have.",
        "Hesitation lets gold go rusty.",
        "This is exactly how I wanted it.",
        "The brighter the risk, the sooner I reach for it.",
        "Fate dealt it. I am not going to refuse."
      ];

      const calm = [
        "Cash flow first. Ambition after.",
        "The ledger is more honest than applause.",
        "No hurry. The advantage will come to me.",
        "Restraint is its own kind of acquisition.",
        "I let the numbers stay polite on my behalf."
      ];

      const sharp = [
        "Some people spend. Others position.",
        "Do not mistake luck for skill.",
        "This table is getting interesting.",
        "I will remember this one.",
        "The spectacle is yours. The profit is mine."
      ];

      let pool = elegant.concat(sharp);
      if (/收藏|月桂|玫瑰|冒险|冲动|火|烈|红|laurel|rose|collector|bold|adventur|impulse|impulsive|fire|flame|red/i.test(name)) pool = bold.concat(elegant);
      if (/银行|黑曜|冷静|理性|账|雪|冰|白|bank|obsidian|calm|cool|rational|restrain|ledger|snow|ice|white/i.test(name)) pool = calm.concat(sharp);

      if (scene === "skip") pool = calm.concat(sharp);
      if (scene === "buy") pool = bold.concat(elegant);
      if (scene === "pay rent" || scene === "pay tax") pool = sharp.concat(calm);
      if (scene === "collect rent" || scene === "bonus") pool = elegant.concat(sharp);
      if (scene === "chance event") pool = bold.concat(sharp, calm);

      const used = getRecentLines(p);
      const available = pool.filter(line => !used.includes(line));
      const choices = available.length ? available : pool;

      return choices[Math.floor(Math.random() * choices.length)];
    }

    function parseJson(text) {
      try { return JSON.parse(text); } catch (e) {}
      const match = String(text || "").match(/\{[\s\S]*\}/);
      if (match) {
        try { return JSON.parse(match[0]); } catch (e) {}
      }
      return {};
    }

    function buyProperty(p, index) {
      const cell = board[index];
      const playerIndex = state.players.indexOf(p);
      if (!cell.price || state.owners[index] !== null || p.money < cell.price) return false;
      p.money -= cell.price;
      p.props.push(index);
      state.owners[index] = playerIndex;
      return true;
    }

    async function playerBuy(buy) {
      const choice = state.awaitingChoice;
      if (!choice) return;

      const p = currentPlayer();
      const cell = board[choice.index];

      if (buy && buyProperty(p, choice.index)) {
        addLog(`✨ ${p.name} buys ${cell.name}.`);
      } else {
        addLog(`🕯️ ${p.name} passes on ${cell.name}.`);
      }

      state.awaitingChoice = null;
      $("choiceModal").classList.add("hidden");
      render();

      if (!state.ended) nextTurn();
    }

    function nextTurn() {
      if (alivePlayers().length <= 1) {
        endGame("Only one player left at the table");
        return;
      }

      const previousIndex = state.currentIndex;
      do {
        state.currentIndex = (state.currentIndex + 1) % state.players.length;
      } while (state.players[state.currentIndex].bankrupt);

      if (state.currentIndex <= previousIndex) {
        state.round += 1;
        if (state.round > state.maxRounds) {
          endGame("Rounds complete");
          return;
        }
      }

      render();
      saveGame();
      maybeAutoTurn();
    }

    function maybeAutoTurn() {
      const p = currentPlayer();
      if (!p || p.type !== "role" || state.ended || state.awaitingChoice) return;
      setTimeout(() => rollCurrent(), 650);
    }

    async function checkBankrupt(p) {
      if (p.money >= 0) return;

      p.bankrupt = true;
      const playerIndex = state.players.indexOf(p);
      p.props.forEach(index => {
        if (state.owners[index] === playerIndex) state.owners[index] = null;
      });
      p.props = [];
      addLog(`🌑 ${p.name} runs out of money and leaves the table.`);

      if (p.type === "role") {
        const line = await roleLine(
          p,
          "bankrupt",
          "You are out of money and about to leave the table. Give one parting line, in character."
        );
        addLog(`${p.name}：「${line}」`);
      }

      if (alivePlayers().length <= 1) await endGame(`${p.name} went bankrupt`);
    }

    async function endGame(reason) {
      if (state.ended) return;

      state.ended = true;
      const ranking = state.players.slice().sort((a, b) => asset(b) - asset(a));
      const winner = ranking[0];
      const detail = ranking.map((p, i) => `${i + 1}. ${p.name} ${money(asset(p))}`).join("；");
      const text = `${reason}. Final standings: ${detail}. The winner is ${winner.name}.`;

      $("endTitle").textContent = winner.type === "human" ? "You take the Star Ring" : `${winner.name} takes the crown`;
      $("endText").textContent = text;
      $("endModal").classList.remove("hidden");
      addLog(`👑 ${text}`);

      if (winner.type === "role") {
        const line = await roleLine(
          winner,
          "victory",
          `You have won the game. Final standings: ${detail}. Give one victory line, in character.`
        );
        addLog(`${winner.name}：「${line}」`);
      }

      render();

      if (!state.demo && API && API.recordGameEvent) {
        try {
          const ids = state.players.filter(p => p.type === "role").map(p => p.id);
          await API.recordGameEvent({
            characterIds: ids,
            summary: `${state.players[0].name} sat down to a game of Star Ring Monopoly with ${state.players.filter(p => p.type === "role").map(p => p.name).join(", ")}. Final standings were ${detail}, and ${winner.name} came out on top.`
          });
        } catch (e) {}
      }

      saveGame();
    }

    async function saveGame() {
      if (!API || !API.saveGame || !state.players.length) return;

      const save = {
        version: 5,
        playerProfile: state.playerProfile,
        characters: state.characters,
        selectedCharacters: state.selectedCharacters,
        recentLines: state.recentLines,
        demo: state.demo,
        currentIndex: state.currentIndex,
        round: state.round,
        maxRounds: state.maxRounds,
        ended: state.ended,
        players: state.players,
        owners: state.owners,
        log: state.log
      };

      try { await API.saveGame(save); } catch (e) {}
    }

    async function newGame() {
      const profile = state.playerProfile;
      const characters = state.characters;

      state = emptyState();
      state.playerProfile = profile;
      state.characters = characters;
      state.selectedCharacters = [];

      $("endModal").classList.add("hidden");
      $("choiceModal").classList.add("hidden");
      $("startModal").classList.remove("hidden");
      $("continueBtn").classList.add("hidden");

      $("startTitle").textContent = "Opening Night";
      $("startDesc").textContent = "Choose 1-3 characters to join the table. Whoever holds the most assets at the end of round 20 wins; anyone who goes bankrupt leaves early.";
      $("startBtn").textContent = "Choose characters to begin";
      $("dice").textContent = "✦";
      $("centerTitle").textContent = "Waiting to begin";
      $("centerText").textContent = "Bring 1-3 characters to the table. Each decides what to buy, and comments and celebrates, according to their own character.";
      $("log").innerHTML = "";

      addLog("✨ Ready for a new game.");
      renderRoles();
      render();
    }

    function restartSamePlayers() {
      setupPlayers();
      $("endModal").classList.add("hidden");
      $("choiceModal").classList.add("hidden");
      $("dice").textContent = "✦";
      $("centerTitle").textContent = "New game";
      $("centerText").textContent = "The Star Ring reshuffles. Good luck.";
      addLog("✨ A new game begins.");
      render();
      saveGame();
      maybeAutoTurn();
    }

    $("startBtn").onclick = () => startGame(false);
    $("continueBtn").onclick = continueSave;
    $("demoBtn").onclick = () => startGame(true);
    $("rollBtn").onclick = () => rollCurrent();
    $("saveBtn").onclick = async () => { await saveGame(); toast("Saved"); };
    $("newBtn").onclick = newGame;
    $("buyBtn").onclick = () => playerBuy(true);
    $("skipBtn").onclick = () => playerBuy(false);
    $("restartBtn").onclick = restartSamePlayers;
    $("closeBtn").onclick = async () => {
      if (API && API.closeGame) await API.closeGame();
      else toast("This environment cannot close the game");
    };

    initHost();
  </script>
